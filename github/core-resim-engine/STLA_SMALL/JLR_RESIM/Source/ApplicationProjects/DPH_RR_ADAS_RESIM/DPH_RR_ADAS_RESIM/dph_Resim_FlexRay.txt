int ReadArguments(int argc, char* argv[])
{
	int count = 1;

	{
		usage();
		return -1;
	}

	if (argc == 2)
	{
		do
		{
			if (argc > count)
			{
				lstrcpy(g_fileName, argv[count]);

			}
			++count;

		} while (0);
	}
	else
	{
		usage();
		return -1;
	}

	return 0;
}

	FR_Data_Obj.detection_rr_radar.dets_info.LookType = c_audi_fr_data.hr.dets.det_info.LookType;
	FR_Data_Obj.detection_rr_radar.dets_info.ScanIndex = c_audi_fr_data.hr.dets.det_info.ScanIndex;

	for(int i = 0 ; i < NUMBER_OF_DETECTIONS_PER_SRR3;i++ ){
		FR_Data_Obj.detection_rr_radar.dets_input[i].amplitude = c_audi_fr_data.hr.dets.det[i].amplitude;
		FR_Data_Obj.detection_rr_radar.dets_input[i].azimuth_raw = c_audi_fr_data.hr.dets.det[i].azimuth_raw;

void FillVehData(SIL_Data_Input_T *pVeh_Track_Obj, char *Line){
	const unsigned int MAX_MEMBERS = 35; // Max required are 28 members for Veh data.
	char param[100] = {0};
	unsigned int i = 0, j = 0,LineLen = strlen(Line);
	uint8 Index = 0;
	float fVEHparam[MAX_MEMBERS] = {0.0};
	do{
		j = 0;
		memset(param, 0, sizeof(param));
		if( Line[i] == '\n') break;
		while(Line[i] != ','){
			param[j] = Line[i];
			i++;
			j++;
			if(i >= LineLen) break;
		}
		param[j]  = '\0';
		fVEHparam[Index] = (float)(atof(param));
		Index++;
		i++;
		if(i >= LineLen) break;
	}while(1);

e_dph_rr_adas_resim_T Veh_Trk_Resim_Run(char* fLog)
{
	std::string csvLine_Veh, csvLine_Trk_RL, csvLine_Trk_RR, str_ch;
	e_dph_rr_adas_resim_T ret = DPH_RR_ADAS_RESIM_NOK;
	SIL_SRR3_AUDI_XML_CONFIG_T *psil_srr3_audi_xml_config = 0;
	SIL_SRR3_AUDI_XML_CONFIG_READ_RESULT sil_srr3_audi_xml_config_result = SIL_SRR3_AUDI_XML_CONFIG_READ_RESULT_ERROR;
	printf("\n[INFO]: Input file name:%s\n", fLog);
	{
		// Test 
		DPH_RR_ADAS_SIL_OUTPUT_TYPES::DPH_SRR3_SIL_VERSION_T ver = {0};
 		m_pSILPlugin->SIL_SRR3_get_version(&ver);

		if (m_pSILPlugin->SIL_SRR3_initialize())
		{
			ret = DPH_RR_ADAS_SIL_NOT_FOUND;
			return ret;
		}
		psil_srr3_audi_xml_config = new SIL_SRR3_AUDI_XML_CONFIG_T;
		psil_srr3_audi_xml_config->xml_config_file_path = xml_file_path;
		m_pSILPlugin->SIL_SRR3_configure(*psil_srr3_audi_xml_config, &sil_srr3_audi_xml_config_result);
		if(sil_srr3_audi_xml_config_result == SIL_SRR3_AUDI_XML_CONFIG_READ_RESULT_ERROR){
			return DPH_RR_ADAS_CONFIG_ERROR;
		}
	}
	Get_All_CSV_fileNames(fLog);

	inputfile_Veh.open(csv_Veh);
	inputfile_Trk_RL.open(csv_RL);
	inputfile_Trk_RR.open(csv_RR);

	
	m_pSILPlugin->SIL_SRR3_Set_VerboseBaseName(fLog);
	
	if (inputfile_Veh.good() && inputfile_Trk_RL.good() & inputfile_Trk_RR.good()){
		
		getline(inputfile_Veh, csvLine_Veh);
		getline(inputfile_Trk_RL, csvLine_Trk_RL);
		getline(inputfile_Trk_RR, csvLine_Trk_RR);

		while(inputfile_Veh.good() && inputfile_Trk_RL.good() & inputfile_Trk_RR.good()){
			
			SIL_Data_Input_T CSV_Veh_Track_Obj = {0};
			char CSVLine[1000] = {0};
			getline(inputfile_Veh, csvLine_Veh);
			strcpy(CSVLine, csvLine_Veh.c_str());
			FillVehData(&CSV_Veh_Track_Obj, CSVLine);
			for(unsigned int i = 0; i < ObjCount; i++){
				getline(inputfile_Trk_RL, csvLine_Trk_RL);
				getline(inputfile_Trk_RR, csvLine_Trk_RR);
				FillTrackerData(&CSV_Veh_Track_Obj, i);
			}

			//Rear left Resim
			CSV_Veh_Track_Obj.Radar_Position = SRR3_RL + 1;
		//	m_pSILPlugin->SIL_SRR3_DATA_Frame_process(&CSV_Veh_Track_Obj, &m_sil_audi_output);

			//Rear Right Resim
			CSV_Veh_Track_Obj.Radar_Position = SRR3_RR + 1;
			//m_pSILPlugin->SIL_SRR3_DATA_Frame_process(&CSV_Veh_Track_Obj, &m_sil_audi_output);

			ret = DPH_RR_ADAS_RESIM_OK;

		} 

	}
	else{
		fprintf(stderr,"Error reading DVL log\n");
		return DPH_RR_ADAS_RESIM_NOK;
	}

	inputfile_Veh.close();
	inputfile_Trk_RL.close();
	inputfile_Trk_RR.close();

	printf("\r\n");
	

	if (psil_srr3_audi_xml_config)
	{
		delete psil_srr3_audi_xml_config;
		psil_srr3_audi_xml_config = 0;
	}

	return ret;
}

void fill_log_status_file(char* status,char* log_fname, char*  comments)
{
	time_t now = time(NULL);
	struct tm *t = localtime(&now);
	fprintf(Statusfp,",%d/%d/%d,",t->tm_mon+1,t->tm_mday,t->tm_year+1900);
	fprintf(Statusfp,"%d:%d:%d,",t->tm_hour,t->tm_min,t->tm_sec);
	fprintf(Statusfp,"%s,%s,%s\n",status,log_fname,comments);
}


void close_log_status_file()
{
	if(Statusfp!=NULL) {
		fprintf(Statusfp,"\nResimulation Operation Completed...\n");
		fclose(Statusfp);
		Statusfp = NULL;
	}
}

			}

		printf("Resimulation Started...\n");
		
		std::ifstream inputFile(g_fileName);
	//	std::ifstream inputFile(ListFileName);
		istream_iterator<line> itFname(inputFile);
		istream_iterator<line> end;
		e_dph_rr_adas_resim_T nStatus = DPH_RR_ADAS_RESIM_INIT;

		if(!inputFile)

			perror("[INFO] Error in Reading Log List...\n Exiting Resimulation...\n\n");
			fill_log_status_file("LOG_LIST_FILE_NOT_FOUND",g_fileName,"Log_list file not found");
		} 

		for(; itFname != end; ++itFname)
		{
			lstrcpy(logname,(*itFname).c_str());
			if (Run_Resim_Log(logname)){
				f_ResimCmplt = 0;
				break;
			}
		}	

	}while(0);
	FreePlugin();
	close_log_status_file();
	if(f_ResimCmplt)

	printf("\nResimulation Operation stopped...");
